# 密文（39字节）
encrypted = [
    0x0D, 0x26, 0x49, 0x45, 0x2A, 0x17, 0x78, 0x44, 0x2B, 0x6C,
    0x5D, 0x5E, 0x45, 0x12, 0x2F, 0x17, 0x2B, 0x44, 0x6F, 0x6E,
    0x56, 0x09, 0x5F, 0x45, 0x47, 0x73, 0x26, 0x0A, 0x0D, 0x13,
    0x17, 0x48, 0x42, 0x01, 0x40, 0x4D, 0x0C, 0x02, 0x69
]

# 初始值
v4 = 4

# 解密
decrypted = [0] * 39

# 从最后一个字符开始
for i in range(38, -1, -1):
    decrypted[i] = v4 ^ encrypted[i]
    v4 = decrypted[i]

# 转换为字符串
flag = ''.join(chr(c) for c in decrypted)
print(f"Flag: {flag}")

# 打印十六进制查看
print("\nHex values:")
print(' '.join(f'0x{c:02X}' for c in decrypted))
